## 第 1 章：核心概念对照

建立 Java 到 Python 的异常映射。throw 对应 raise，try-catch 对应 try-except，@ControllerAdvice 对应 @shturl.cc/NiEAw9PB0sm。两者思想完全一致，只是关键字和注册方式不同，有 Java 基础可快速迁移。

```python
# Java: throw new RuntimeException("错误")
# Python:
raise ValueError("错误")

# Java: try { ... } catch (Exception e) { ... } finally { ... }
# Python:
try:
    risky_operation()
except ValueError as e:
    print(f"捕获异常: {e}")
finally:
    print("始终执行")
```

| Spring Boot               | FastAPI                  | 说明              |
| :------------------------ | :----------------------- | :---------------- |
| `@ControllerAdvice`       | `@app.exception_handler` | 全局异常处理      |
| `ResponseStatusException` | `HTTPException`          | 框架内置HTTP异常  |
| 自定义 `RuntimeException` | 自定义 `Exception`       | 业务异常          |
| `@ExceptionHandler`       | `@app.exception_handler` | 注册异常处理器    |
| `@Valid` 校验失败 → 400   | `RequestValidationError` | Pydantic 校验异常 |
| `try-catch`               | `try-except`             | 捕获异常          |
| `finally`                 | `finally`                | 最终执行          |
| `throw`                   | `raise`                  | 抛出异常          |

## 第 2 章：自定义异常体系

创建以 AppException 为基类的业务异常树，包含 code、message、status_code 三个属性。子类如 NotFoundException、ConflictException、ForbiddenException 只传具体参数，让异常类型本身表达业务语义，无需重复定义状态码。

```python
# app/core/exceptions.py

class AppException(Exception):
    """业务异常基类"""
    def __init__(self, code: str, message: str, status_code: int = 400):
        self.code = code
        self.message = message
        self.status_code = status_code
        super().__init__(message)


class NotFoundException(AppException):
    """资源不存在（404）"""
    def __init__(self, resource: str, identifier: str | int):
        super().__init__("NOT_FOUND", f"{resource}不存在: {identifier}", 404)


class ConflictException(AppException):
    """资源冲突（409）"""
    def __init__(self, message: str):
        super().__init__("CONFLICT", message, 409)


class UnauthorizedException(AppException):
    """未认证（401）"""
    def __init__(self, message: str = "未登录或Token已过期"):
        super().__init__("UNAUTHORIZED", message, 401)


class ForbiddenException(AppException):
    """无权限（403）"""
    def __init__(self, message: str = "权限不足"):
        super().__init__("FORBIDDEN", message, 403)
```

## 第 3 章：全局异常处理器

注册四层异常处理器，统一响应格式 code、message、data。自定义业务异常返回对应状态码，Pydantic 校验失败提取字段错误信息，框架 HTTP 异常直接转换，最外层兜底捕获所有未处理异常返回 500。注册顺序从具体到通用。

```python
# app/core/exception_handlers.py
from fastapi import Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from starlette.exceptions import HTTPException as StarletteHTTPException
from app.core.exceptions import AppException
from app.core.logger import logger


async def app_exception_handler(request: Request, exc: AppException):
    """自定义业务异常 → 统一JSON"""
    logger.warning(f"业务异常 [{exc.code}] {exc.message}")
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": exc.code, "message": exc.message, "data": None}
    )


async def http_exception_handler(request: Request, exc: StarletteHTTPException):
    """框架HTTP异常"""
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": "HTTP_ERROR", "message": str(exc.detail), "data": None}
    )


async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """Pydantic校验失败 → 提取字段错误"""
    messages = []
    for error in exc.errors():
        field = ".".join(str(loc) for loc in error["loc"])
        messages.append(f"{field}: {error['msg']}")
    
    logger.warning(f"参数校验失败: {messages}")
    return JSONResponse(
        status_code=422,
        content={"code": "VALIDATION_ERROR", "message": "; ".join(messages), "data": None}
    )


async def global_exception_handler(request: Request, exc: Exception):
    """兜底：未预料异常 → 500"""
    logger.error(f"未处理异常: {type(exc).__name__}: {exc}", exc_info=True)
    return JSONResponse(
        status_code=500,
        content={"code": "INTERNAL_ERROR", "message": "服务器内部错误", "data": None}
    )


# ===== 注册（app/main.py） =====
def register_exception_handlers(app: FastAPI):
    app.add_exception_handler(AppException, app_exception_handler)
    app.add_exception_handler(StarletteHTTPException, http_exception_handler)
    app.add_exception_handler(RequestValidationError, validation_exception_handler)
    app.add_exception_handler(Exception, global_exception_handler)
```

## 第 4 章：Service 层异常实践

业务层只抛业务异常不处理 HTTP 细节。查不到资源抛 NotFoundException，数据冲突抛 ConflictException。异常自动向上传播到全局处理器，Service 代码干净无需 try-except 转换，职责单一，只关注业务逻辑。

```python
# app/services/user_service.py
from app.core.exceptions import NotFoundException, ConflictException

class UserService:
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def get_by_id(self, user_id: int) -> User:
        """查用户，不存在抛异常"""
        user = await self.db.get(User, user_id)
        if not user:
            raise NotFoundException("用户", user_id)
        return user
    
    async def create(self, data: UserCreate) -> User:
        """创建用户，冲突抛异常"""
        existing = await self.db.scalar(
            select(User).where(User.username == data.username)
        )
        if existing:
            raise ConflictException(f"用户名已存在: {data.username}")
        
        user = User(**data.model_dump())
        self.db.add(user)
        await self.db.flush()
        return user
    
    async def update(self, user_id: int, data: UserUpdate) -> User:
        """更新用户"""
        user = await self.get_by_id(user_id)  # 不存在自动抛404
        
        if data.username:
            existing = await self.db.scalar(
                select(User).where(User.username == data.username, User.id != user_id)
            )
            if existing:
                raise ConflictException(f"用户名已被占用: {data.username}")
        
        update_data = data.model_dump(exclude_unset=True)
        for field, value in update_data.items():
            setattr(user, field, value)
        
        await self.db.flush()
        return user
    
    async def delete(self, user_id: int) -> None:
        """删除用户"""
        user = await self.get_by_id(user_id)
        await self.db.delete(user)
        await self.db.flush()
```

## 第 5 章：路由层自动传播

路由函数不写 try-except，Pydantic 自动校验请求体失败抛 422，Service 抛出的业务异常自动穿过路由层到达全局处理器。路由代码只关注正常业务流程，异常处理完全解耦，代码简洁可读。

```python
# app/api/v1/users.py
from fastapi import APIRouter, Depends

router = APIRouter(prefix="/api/v1/users", tags=["用户管理"])

@router.get("/{user_id}")
async def get_user(
    user_id: int,
    service: UserService = Depends(get_user_service)
):
    """异常自动传播，无需 try-except"""
    return await service.get_by_id(user_id)

@router.post("/")
async def create_user(
    data: UserCreate,  # Pydantic 自动校验，失败抛422
    service: UserService = Depends(get_user_service)
):
    return await service.create(data)

@router.patch("/{user_id}")
async def update_user(
    user_id: int,
    data: UserUpdate,
    service: UserService = Depends(get_user_service)
):
    return await service.update(user_id, data)

@router.delete("/{user_id}", status_code=204)
async def delete_user(
    user_id: int,
    service: UserService = Depends(get_user_service)
):
    await service.delete(user_id)
    return None
```

## 第 6 章：数据库异常处理

在 Service 层捕获 SQLAlchemy 特有异常并转换。IntegrityError 唯一约束冲突转为 ConflictException，NoResultFound 转为 NotFoundException，OperationalError 数据库连接失败转为 500 并记录日志，避免暴露数据库细节给前端。

```python
# app/services/user_service.py 补充
from sqlalchemy.exc import IntegrityError, OperationalError

class UserService:
    
    async def create_safe(self, data: UserCreate) -> User:
        """创建用户，处理数据库异常"""
        try:
            user = User(**data.model_dump())
            self.db.add(user)
            await self.db.flush()
            return user
        except IntegrityError as e:
            await self.db.rollback()
            if "username" in str(e.orig):
                raise ConflictException(f"用户名已存在: {data.username}")
            if "email" in str(e.orig):
                raise ConflictException(f"邮箱已被注册: {data.email}")
            raise ConflictException("数据冲突")
        except OperationalError as e:
            await self.db.rollback()
            logger.error(f"数据库连接失败: {e}", exc_info=True)
            raise AppException("DB_ERROR", "数据库服务异常", 500)
```

## 第 7 章：日志系统概述

Python 推荐用 loguru，比内置 logging 简洁十倍。日志和异常是一体的，异常被捕获时同步记录日志，形成完整链路。支持控制台彩色输出、按大小时间切割文件、自动归档过期日志、结构化绑定上下文信息。

```python
# 安装
# pip install loguru

# 对比内置 logging（不用这个）
import logging
logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
logging.info("内置logging，配置繁琐")

# loguru（推荐用这个）
from loguru import logger
logger.info("开箱即用，无需配置")
logger.error(f"自动格式化: {value}", exc_info=True)  # 一行记录堆栈
```

## 第 8 章：日志级别与使用场景

DEBUG 用于开发调试打印变量值和 SQL，INFO 记录关键业务流程节点如登录成功，WARNING 记录业务异常和降级处理，ERROR 记录系统错误需要人工介入，必须附带堆栈，CRITICAL 记录系统崩溃或数据丢失风险，需立即告警。

```python
from loguru import logger

# DEBUG：开发调试
logger.debug(f"SQL: {stmt}")
logger.debug(f"请求参数: {request.query_params}")

# INFO：关键业务流程
logger.info(f"用户登录成功: user_id={user.id}")
logger.info(f"订单创建: order_no={order.order_no}")

# WARNING：业务异常、降级
logger.warning(f"用户名已存在: {data.username}")
logger.warning(f"缓存未命中，降级查库: key={cache_key}")

# ERROR：系统错误，必须带堆栈
try:
    risky_operation()
except Exception as e:
    logger.error(f"数据库连接失败: {e}", exc_info=True)

# CRITICAL：系统崩溃
logger.critical(f"Redis集群全部宕机，缓存全量失效")
```

## 第 9 章：日志配置与使用

移除默认配置后分别添加控制台和文件输出。控制台用彩色格式便于开发调试，文件按 10MB 切割保留 30 天。在全局异常处理器中记录不同级别日志，业务异常用 warning，系统异常用 error 并附带 exc_info=True 记录完整堆栈。

```python
# app/core/logger.py
import sys
from pathlib import Path
from loguru import logger

# 移除默认配置
logger.remove()

# 控制台输出（开发环境，彩色）
logger.add(
    sys.stderr,
    format="<green>{time:HH:mm:ss}</green> | <level>{level: <8}</level> | <level>{message}</level>",
    level="DEBUG",
    colorize=True
)

# 文件输出（生产环境，按大小切割）
logger.add(
    "logs/app_{time:YYYY-MM-DD}.log",
    rotation="10 MB",        # 10MB 切割
    retention="30 days",     # 保留30天
    compression="zip",       # 压缩归档
    level="INFO",
    format="{time:YYYY-MM-DD HH:mm:ss} | {level} | {name}:{function}:{line} | {message} | request_id={extra[request_id]}"
)

# 错误日志单独文件
logger.add(
    "logs/error_{time:YYYY-MM-DD}.log",
    rotation="10 MB",
    retention="90 days",
    level="ERROR",
    format="{time:YYYY-MM-DD HH:mm:ss} | {level} | {message} | {exception}"
)
```

## 第 10 章：请求追踪

用中间件给每个请求分配唯一 request_id，通过 shturl.cc/G 绑定到日志上下文。整个请求链路的所有日志自动携带该 ID，跨 Service、数据库、异常处理器追踪完整调用链，类似 Java 的 MDC 机制，排查问题只需一个 ID。

```python
# app/middleware/request_id.py
import uuid
from fastapi import Request
from loguru import logger
from starlette.middleware.base import BaseHTTPMiddleware


class RequestIDMiddleware(BaseHTTPMiddleware):
    """给每个请求分配唯一ID，绑定到日志上下文"""
    
    async def dispatch(self, request: Request, call_next):
        # 优先使用客户端传入的，否则生成新的
        request_id = request.headers.get("X-Request-ID", str(uuid.uuid4()))
        request.state.request_id = request_id
        
        # 绑定到日志上下文（类似 Java MDC）
        with logger.contextualize(request_id=request_id):
            logger.info(f"→ {request.method} {request.url.path}")
            response = await call_next(request)
            response.headers["X-Request-ID"] = request_id
            logger.info(f"← {response.status_code}")
            return response


# app/main.py 注册
app.add_middleware(RequestIDMiddleware)
```

## 第 11 章：生产环境最佳实践

区分开发与生产环境的异常和日志策略。开发环境返回详细错误加堆栈，生产环境返回通用消息隐藏内部细节。业务异常记 warning 日志，系统异常记 error 日志。确保任何未处理异常都返回 JSON 而非 HTML，日志文件定期清理避免撑满磁盘，敏感信息不记录到日志。

```python
# app/core/exception_handlers.py 补充
from app.core.config import settings

async def global_exception_handler(request: Request, exc: Exception):
    """生产环境隐藏错误细节"""
    logger.error(f"未处理异常: {type(exc).__name__}: {exc}", exc_info=True)
    
    # 开发环境：返回详细错误
    if settings.DEBUG:
        return JSONResponse(
            status_code=500,
            content={
                "code": "INTERNAL_ERROR",
                "message": str(exc),
                "traceback": traceback.format_exc()  # 开发用
            }
        )
    
    # 生产环境：返回通用消息
    return JSONResponse(
        status_code=500,
        content={
            "code": "INTERNAL_ERROR",
            "message": "服务器内部错误，请稍后重试",
            "data": None
        }
    )


# 敏感信息脱敏
def mask_sensitive(data: dict) -> dict:
    """脱敏手机号、密码等"""
    if "password" in data:
        data["password"] = "***"
    if "phone" in data:
        data["phone"] = data["phone"][:3] + "****" + data["phone"][-4:]
    return data

# 日志中使用
logger.info(f"用户注册: {mask_sensitive(request_data)}")
```
